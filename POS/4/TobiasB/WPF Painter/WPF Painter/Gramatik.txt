<Block> ::= { <Program> }
<Program> ::= <Statement> { <Statement> }
<Statement> ::= <DRAW> | <TURN> | <COLOR> | <FOR>
<FOR> ::= FOR <Number> <Block>
<DRAW> ::= DRAW <Number>
<TURN> ::= TURN <Direction> <Number> | TURN <Direction> <Number>
<COLOR> ::= COLOR <ColorType>
<Direction> ::= RIGHT | LEFT | UP | DOWN
<ColorType> ::= RED | BLUE | YELLOW | GREEN | … // kann man auch als mehrere Letter machen A-Z a-z
<Number> ::= 1 | 2 | 3 | 4 | ...
